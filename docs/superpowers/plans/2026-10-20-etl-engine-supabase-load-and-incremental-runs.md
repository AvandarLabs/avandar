# EtlEngine Supabase Load and Incremental Runs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `@avandar/etl` own the Supabase load step for every pipeline, and give every pipeline resumable, per-drive incremental runs driven by the source's modification time.

**Architecture:** A new generic `EtlEngine.create(config)` crawls each drive oldest-first, writes one page at a time through a `ListingsStore` and an `ObjectStore`, and resumes from `max(date_of_last_update)` per pipeline and drive. Pipelines supply only `listDrives`, `extract`, and `transform`. HDX moves to keyset-paged `package_search`, Socrata to `order=updatedAt ASC`, and `world-bank__wdi` to engine-owned Parquet loading.

**Tech Stack:** TypeScript, Node 22, Vitest, `@supabase/supabase-js`, DuckDB (`duckdb` Node bindings), Fastify, Acclimate (`ava` CLI), Postgres + pgTAP, CKAN `package_search`, Socrata Discovery API.

**Spec:** [`docs/superpowers/specs/2026-10-20-etl-engine-supabase-load-and-incremental-runs-design.md`](../specs/2026-10-20-etl-engine-supabase-load-and-incremental-runs-design.md).
Test ids (E1, L4, S3, ...) below refer to the spec's section 8.

```ts
import { randomUUID } from "node:crypto";
import { buildListingKeyFromRow } from "@etl/stores/buildListingKeyFromRow/buildListingKeyFromRow";
import type {
  ListingRowValues,
  ListingsStore,
  RelationColumnRow,
  StoredListingRow,
} from "@etl/stores/EtlStores.types";

/** A listings store held in memory, plus accessors for asserting on it. */
export type InMemoryListingsStore = ListingsStore &
  Readonly<{
    /** Every stored row, in the order its key was first written. */
    getRows: () => readonly StoredListingRow[];
    /** One listing's relation columns, as last replaced. */
    getRelationColumns: (
      listingId: string,
    ) => readonly RelationColumnRow[] | undefined;
    /** How many rows each `readRowsMatching` call asked for, in call order. */
    getReadBatchSizes: () => readonly number[];
    /** Every `upsertRows` payload, in call order. */
    getUpsertBatches: () => readonly (readonly ListingRowValues[])[];
  }>;

function _readHighWaterMark(args: {
  rows: Iterable<StoredListingRow>;
  pipelineName: string;
  apiBaseUrl: string;
}): Date | undefined {
  const newest = [...args.rows].reduce<number | undefined>((max, row) => {
    const isInDrive =
      row.pipeline_name === args.pipelineName &&
      row.api_base_url === args.apiBaseUrl &&
      row.date_of_last_update !== null;
    if (!isInDrive) {
      return max;
    }
    const time = Date.parse(row.date_of_last_update ?? "");
    return max === undefined || time > max ? time : max;
  }, undefined);
  return newest === undefined ? undefined : new Date(newest);
}

/**
 * Creates a listings store that keeps rows in memory, for testing a pipeline
 * or the engine without Supabase. Upserts keep a row's id and merge over any
 * columns outside the contract, the way Postgres keeps columns an upsert
 * does not name.
 *
 * @param options.rows - Rows to start with.
 * @returns The store.
 */
export function createInMemoryListingsStore(
  options: Readonly<{ rows?: readonly StoredListingRow[] }> = {},
): InMemoryListingsStore {
  const rowsByKey = new Map<string, StoredListingRow>(
    (options.rows ?? []).map((row) => {
      return [buildListingKeyFromRow(row), row];
    }),
  );
  const columnsByListingId = new Map<string, readonly RelationColumnRow[]>();
  const readBatchSizes: number[] = [];
  const upsertBatches: (readonly ListingRowValues[])[] = [];

  return {
    readHighWaterMark: async ({ pipelineName, apiBaseUrl }) => {
      return _readHighWaterMark({ rows: rowsByKey.values(), pipelineName, apiBaseUrl });
    },
    readRowsMatching: async (rows) => {
      readBatchSizes.push(rows.length);
      return rows.flatMap((row) => {
        const stored = rowsByKey.get(buildListingKeyFromRow(row));
        return stored === undefined ? [] : [stored];
      });
    },
    upsertRows: async (rows) => {
      upsertBatches.push([...rows]);
      return rows.map((row) => {
        const key = buildListingKeyFromRow(row);
        const existing = rowsByKey.get(key);
        const stored = { ...existing, ...row, id: existing?.id ?? randomUUID() };
        rowsByKey.set(key, stored);
        return stored;
      });
    },
    replaceRelationColumns: async ({ listingId, columns }) => {
      columnsByListingId.set(listingId, [...columns]);
    },
    getRows: () => {
      return [...rowsByKey.values()];
    },
    getRelationColumns: (listingId) => {
      return columnsByListingId.get(listingId);
    },
    getReadBatchSizes: () => {
      return [...readBatchSizes];
    },
    getUpsertBatches: () => {
      return [...upsertBatches];
    },
  };
}
```

```ts
import type {
  EtlRunOptions,
  EtlRunWindow,
} from "@etl/EtlEngine/EtlEngine.types";

/**
 * Resolves one drive's crawl window. An explicit `since` wins. Otherwise an
 * incremental run starts at the drive's high-water mark, and a full run has
 * no lower bound. `until` always passes through.
 *
 * @param args.options - The run's options.
 * @param args.highWaterMark - The drive's high-water mark, if it was read.
 * @returns The window.
 */
export function resolveRunWindow(args: {
  options: EtlRunOptions;
  highWaterMark: Date | undefined;
}): EtlRunWindow {
  const { options, highWaterMark } = args;
  const since =
    options.since ??
    (options.mode === "incremental" ? highWaterMark : undefined);
  return { since, until: options.until };
}
```

```ts
import { LISTING_CONTRACT_COLUMNS } from "@etl/stores/listingContractColumns";
import type {
  ListingRowValues,
  StoredListingRow,
} from "@etl/stores/EtlStores.types";

/** Columns that change on every write, so they never count as a change. */
const IGNORED_COLUMNS: ReadonlySet<keyof ListingRowValues> = new Set([
  "pipeline_run_id",
  "date_of_last_sync",
]);

/** Columns compared as instants: Postgres returns `+00:00`, not `Z`. */
const TIMESTAMP_COLUMNS: ReadonlySet<keyof ListingRowValues> = new Set([
  "date_of_last_update",
  "coverage_start_date",
  "coverage_end_date",
]);

function _stableStringify(value: unknown): string {
  if (Array.isArray(value)) {
    return `[${value.map(_stableStringify).join(",")}]`;
  }
  if (typeof value === "object" && value !== null) {
    const entries = Object.entries(value).sort(([keyA], [keyB]) => {
      return keyA < keyB ? -1 : 1;
    });
    const body = entries.map(([key, entryValue]) => {
      return `${JSON.stringify(key)}:${_stableStringify(entryValue)}`;
    });
    return `{${body.join(",")}}`;
  }
  return JSON.stringify(value);
}

function _normalize(column: keyof ListingRowValues, value: unknown): string {
  if (value === null || value === undefined) {
    return "null";
  }
  if (TIMESTAMP_COLUMNS.has(column)) {
    return String(Date.parse(String(value)));
  }
  return _stableStringify(value);
}

/**
 * Whether writing `next` would change nothing that matters about `stored`.
 * Every contract column is compared except `pipeline_run_id` and
 * `date_of_last_sync`, so a change to a pipeline's own mapping is caught
 * even when the source did not change.
 *
 * @param args.stored - The row as the listings table holds it.
 * @param args.next - The row this run would write.
 * @returns `true` when the write can be skipped.
 */
export function isListingRowUnchanged(args: {
  stored: StoredListingRow;
  next: ListingRowValues;
}): boolean {
  return LISTING_CONTRACT_COLUMNS.every((column) => {
    if (IGNORED_COLUMNS.has(column)) {
      return true;
    }
    return (
      _normalize(column, args.stored[column]) ===
      _normalize(column, args.next[column])
    );
  });
}
```

```ts
      columnNames: summary.columnNames,
      columnTypes: summary.columnTypeDescriptions,
    };
  } finally {
    await db.close();
  }
}

async function _buildParquetFile(args: {
  relation: PipelineParquetRelation;
  pipelineName: string;
  pipelineRunId: EtlPipelineRunId;
}): Promise<_ParquetFile> {
  const { relation } = args;
  const [blob] = await transformedCsvsToParquetBlobs({
    transformOutputDir: dirname(relation.csvPath),
    descriptions: [
      { name: basename(relation.csvPath, ".csv"), columns: relation.csvColumns },
    ],
  });
  if (blob === undefined) {
    throw new Error(`No Parquet was produced for "${relation.tableBaseName}".`);
  }
  const bytes = new Uint8Array(await blob.arrayBuffer());
  const loadDir = getEtlLoadDir(args.pipelineName, args.pipelineRunId);
  await mkdir(loadDir, { recursive: true });
  const parquetPath = join(loadDir, `${relation.tableBaseName}.parquet`);
  await writeFile(parquetPath, bytes);
  return {
    bytes,
    sha256: createHash("sha256").update(bytes).digest("hex"),
    ...(await _summarizeParquetFile(parquetPath)),
  };
}

function _buildColumnRows(parquet: _ParquetFile): RelationColumnRow[] {
  return parquet.columnNames.map((columnName, index) => {
    const columnType = parquet.columnTypes[index] ?? "VARCHAR";
    return {
      column_name: columnName,
      display_order: index,
      original_data_type: columnType,
      cast_data_type: duckDbDescribeColumnTypeToSniffable(columnType),
    };
  });
}

/**
 * Converts a Parquet relation's CSV to ZSTD Parquet, uploads it unless the
 * stored row already records the same SHA-256, and returns the row with the
 * engine's Parquet metadata (`open_data_bucket`, `storage_object_path`,
 * `table`, `parquet_sha256`) merged over the relation's own and the object's
 * public URL appended to `canonical_urls`. The columns come from the Parquet
 * file's `DESCRIBE`.
 *
 * @throws When no bucket is configured.
 */
export async function prepareParquetRelation(args: {
  relation: PipelineParquetRelation;
  row: ListingRowValues;
  stored: StoredListingRow | undefined;
  pipelineName: string;
  pipelineRunId: EtlPipelineRunId;
  objects: ObjectStore | undefined;
  bucket: string | undefined;
}): Promise<PreparedParquetRelation> {
  const { relation, row, objects, bucket } = args;
  if (objects === undefined || bucket === undefined) {
    throw new Error(
      `EtlEngine needs output.bucket to write the pipeline_parquet relation "${relation.tableBaseName}".`,
    );
  }
  const parquet = await _buildParquetFile(args);
  const storagePath = `${args.pipelineName}/datasets/${relation.tableBaseName}.parquet`;
  if (args.stored?.metadata?.parquet_sha256 !== parquet.sha256) {
    await objects.upload({ path: storagePath, bytes: parquet.bytes, contentType: PARQUET_CONTENT_TYPE });
  }
  const metadata = {
    ...(row.metadata ?? {}),
    open_data_bucket: bucket,
    storage_object_path: storagePath,
    table: { name: relation.tableBaseName, row_count: parquet.rowCount, column_names: [...parquet.columnNames] },
    parquet_sha256: parquet.sha256,
  };
  const canonicalUrls = [...(row.canonical_urls ?? []), objects.getPublicUrl(storagePath)];
  return { row: { ...row, metadata, canonical_urls: canonicalUrls }, columns: _buildColumnRows(parquet) };
}
```

```ts
import { buildListingRowFromRelation } from "@etl/EtlEngine/buildListingRowFromRelation/buildListingRowFromRelation";
import { isListingRowUnchanged } from "@etl/EtlEngine/isListingRowUnchanged/isListingRowUnchanged";

import { buildListingKeyFromRow } from "@etl/stores/buildListingKeyFromRow/buildListingKeyFromRow";
import type {
  EtlPipelineRunId,
  EtlStores,
  Relation,
  RelationColumn,

  ListingRowValues,

/** Rows per read and per upsert, so each request's URL and body stay small. */
export const LISTING_CHUNK_SIZE = 200;

/** What `loadRelationsPage` needs to know about the run and drive. */
export type LoadRelationsPageContext = Readonly<{
  pipelineName: string;
  pipelineRunId: EtlPipelineRunId;
  service: string | undefined;
  driveKey: string;
  syncedAt: Date;
  stores: EtlStores;
  bucket: string | undefined;
}>;

/** How many of a page's relations were written and how many were skipped. */
export type LoadRelationsPageResult = Readonly<{
  written: number;
  unchanged: number;
}>;

type _PreparedRelation = Readonly<{
  row: ListingRowValues;
  columns: readonly RelationColumnRow[] | undefined;
}>;

function _chunk<T>(items: readonly T[], size: number): T[][] {
  return Array.from({ length: Math.ceil(items.length / size) }, (_, index) => {
    return items.slice(index * size, (index + 1) * size);
  });
}

function _toColumnRow(column: RelationColumn, index: number): RelationColumnRow {
  return {
    column_name: column.name,
    display_order: index,
    original_data_type: column.originalDataType,
    cast_data_type: column.castDataType,
  };
}

async function _readStoredRowsByKey(args: {
  rows: readonly ListingRowValues[];
  listings: ListingsStore;
}): Promise<ReadonlyMap<string, StoredListingRow>> {
  const chunks = await promiseMapSequential(
    _chunk(args.rows, LISTING_CHUNK_SIZE),
    (chunk) => {
      return args.listings.readRowsMatching(chunk);
    },
  );
  return new Map(
    chunks.flat().map((stored) => {
      return [buildListingKeyFromRow(stored), stored];
    }),
  );
}

async function _prepareRelation(args: {
  relation: Relation;
  row: ListingRowValues;
  stored: StoredListingRow | undefined;
  context: LoadRelationsPageContext;
}): Promise<_PreparedRelation> {
  const { relation, context } = args;
  if (relation.kind === "pipeline_parquet") {
    return prepareParquetRelation({
      relation,
      row: args.row,
      stored: args.stored,
      pipelineName: context.pipelineName,
      pipelineRunId: context.pipelineRunId,
      objects: context.stores.objects,
      bucket: context.bucket,
    });
  }
  return { row: args.row, columns: relation.columns?.map(_toColumnRow) };
}

async function _writeChanged(args: {
  changed: readonly _PreparedRelation[];
  listings: ListingsStore;
}): Promise<void> {
  const chunks = _chunk(args.changed, LISTING_CHUNK_SIZE);
  // Sequential on purpose: chunks go oldest connection first, so a failure
  // partway leaves an oldest-first prefix of the page written.
  await promiseMapSequential(chunks, async (chunk) => {
    const written = await args.listings.upsertRows(
      chunk.map((prepared) => {
        return prepared.row;
      }),
    );
    const idByKey = new Map(
      written.map((stored) => {
        return [buildListingKeyFromRow(stored), stored.id];
      }),
    );
    await promiseMapSequential(chunk, async ({ row, columns }) => {
      const listingId = idByKey.get(buildListingKeyFromRow(row));
      if (columns === undefined || listingId === undefined) {
        return;
      }
      await args.listings.replaceRelationColumns({ listingId, columns });
    });
  });
}

/**
 * Loads one page of relations: reads the stored rows for the page, builds
 * any Parquet objects, and writes only the rows that changed, each followed
 * by its relation columns. Relations must be ordered oldest connection first.
 *
 * @param args.relations - The page's relations, oldest connection first.
 * @param args.context - The run and drive the page belongs to.
 * @returns How many relations were written and how many were unchanged.
 */
export async function loadRelationsPage(args: {
  relations: readonly Relation[];
  context: LoadRelationsPageContext;
}): Promise<LoadRelationsPageResult> {
  const { relations, context } = args;
  const pairs = relations.map((relation) => {
    return {
      relation,
      row: buildListingRowFromRelation({
        relation,
        pipelineName: context.pipelineName,
        pipelineRunId: context.pipelineRunId,
        service: context.service,
        driveKey: context.driveKey,
        syncedAt: context.syncedAt,
      }),
    };
  });
  const storedByKey = await _readStoredRowsByKey({
    rows: pairs.map((pair) => {
      return pair.row;
    }),
    listings: context.stores.listings,
  });
  const prepared = await promiseMapSequential(pairs, ({ relation, row }) => {
    return _prepareRelation({ relation, row, stored: storedByKey.get(buildListingKeyFromRow(row)), context });
  });
  const changed = prepared.filter(({ row }) => {
    const stored = storedByKey.get(buildListingKeyFromRow(row));
    return stored === undefined || !isListingRowUnchanged({ stored, next: row });
  });
  await _writeChanged({ changed, listings: context.stores.listings });
  return { written: changed.length, unchanged: prepared.length - changed.length };
}
```

```ts
  seenIds: ReadonlySet<string>;
  /** The timestamp of the last connection seen, when timestamps are read. */
  lastTimestamp: Date | undefined;
  /** How many more connections the run's limit allows. */
  remaining: number;
}>;

function _dropRepeats<TConnection>(args: {
  page: readonly TConnection[];
  seenIds: ReadonlySet<string>;
  getConnectionId: (connection: TConnection) => string;
}): TConnection[] {
  const ids = new Set(args.seenIds);
  return args.page.filter((connection) => {
    const id = args.getConnectionId(connection);
    if (ids.has(id)) {
      return false;
    }
    ids.add(id);
    return true;
  });
}

function _assertOldestFirst<TConnection>(args: {
  connections: readonly TConnection[];
  driveKey: string;
  lastTimestamp: Date | undefined;
  getConnectionId: (connection: TConnection) => string;
  getConnectionTimestamp: (connection: TConnection) => Date;
}): Date | undefined {
  return args.connections.reduce<Date | undefined>((previous, connection) => {
    const timestamp = args.getConnectionTimestamp(connection);
    if (previous !== undefined && timestamp.getTime() < previous.getTime()) {
      throw new OutOfOrderConnectionError({
        driveKey: args.driveKey,
        connectionId: args.getConnectionId(connection),
        timestamp,
        previousTimestamp: previous,
      });
    }
    return timestamp;
  }, args.lastTimestamp);
}

function _isInWindow(timestamp: Date, window: EtlRunWindow): boolean {
  const time = timestamp.getTime();
  return (
    (window.since === undefined || time >= window.since.getTime()) &&
    (window.until === undefined || time < window.until.getTime())
  );
}

/**
 * Picks which of a page's connections to process: repeats are dropped,
 * then (when timestamps are read) the rest must be oldest first and inside
 * the window, then the page is trimmed to the run's remaining limit.
 *
 * @returns The connections to process and the drive's updated state.
 * @throws OutOfOrderConnectionError when a connection is older than the one
 * before it.
 */
export function selectPageConnections<TConnection>(args: {
  page: readonly TConnection[];
  driveKey: string;
  window: EtlRunWindow;
  state: ConnectionSelectionState;
  getConnectionId: (connection: TConnection) => string;
  getConnectionTimestamp: ((connection: TConnection) => Date) | undefined;
}): Readonly<{
  connections: readonly TConnection[];
  state: ConnectionSelectionState;
}> {
  const { state, getConnectionId, getConnectionTimestamp } = args;
  const unseen = _dropRepeats({ page: args.page, seenIds: state.seenIds, getConnectionId });
  const lastTimestamp =
    getConnectionTimestamp === undefined
      ? undefined
      : _assertOldestFirst({ connections: unseen, driveKey: args.driveKey, lastTimestamp: state.lastTimestamp, getConnectionId, getConnectionTimestamp });
  const inWindow =
    getConnectionTimestamp === undefined
      ? unseen
      : unseen.filter((connection) => {
          return _isInWindow(getConnectionTimestamp(connection), args.window);
        });
  const connections = inWindow.slice(0, state.remaining);
  return {
    connections,
    state: {
      seenIds: new Set([...state.seenIds, ...unseen.map(getConnectionId)]),
      lastTimestamp,
      remaining: state.remaining - connections.length,
    },
  };
}
```

```ts
import { buildListingKeyFromRow } from "@etl/stores/buildListingKeyFromRow/buildListingKeyFromRow";
import { LISTING_CONTRACT_COLUMNS } from "@etl/stores/listingContractColumns";
import type {
  ListingRowValues,
  ListingsStore,
  RelationColumnRow,
  StoredListingRow,
} from "@etl/stores/EtlStores.types";
import type { SupabaseClient } from "@supabase/supabase-js";

const SELECT_COLUMNS = ["id", ...LISTING_CONTRACT_COLUMNS].join(",");
const API_RESOURCE_CONFLICT =
  "api_service,api_base_url,external_dataset_id,api_resource_id";
const PIPELINE_PARQUET_CONFLICT = "parquet_file_name,pipeline_name";

type _Tables = Readonly<{
  client: SupabaseClient;
  listingsTable: string;
  relationColumnsTable: string;
}>;

function _throwOnError(args: {
  error: { message: string } | null;
  action: string;
}): void {
  if (args.error) {
    throw new Error(`Supabase failed ${args.action}: ${args.error.message}`);
  }
}

function _groupBy<T>(
  items: readonly T[],
  getKey: (item: T) => string,
): ReadonlyMap<string, readonly T[]> {
  const groups = new Map<string, T[]>();
  items.forEach((item) => {
    const key = getKey(item);
    groups.set(key, [...(groups.get(key) ?? []), item]);
  });
  return groups;
}

function _unique(values: ReadonlyArray<string | null>): string[] {
  return [
    ...new Set(
      values.filter((value): value is string => {
        return value !== null;
      }),
    ),
  ];
}

async function _readHighWaterMark(
  args: _Tables & { pipelineName: string; apiBaseUrl: string },
): Promise<Date | undefined> {
  const { data, error } = await args.client
    .from(args.listingsTable)
    .select("date_of_last_update")
    .eq("pipeline_name", args.pipelineName)
    .eq("api_base_url", args.apiBaseUrl)
    .not("date_of_last_update", "is", null)
    .order("date_of_last_update", { ascending: false })
    .limit(1);
  _throwOnError({ error, action: `reading the high-water mark from ${args.listingsTable}` });
  const rows = (data ?? []) as ReadonlyArray<{ date_of_last_update: string }>;
  const newest = rows[0]?.date_of_last_update;
  return newest === undefined ? undefined : new Date(newest);
}

async function _readGroup(args: {
  tables: _Tables;
  group: readonly ListingRowValues[];
}): Promise<readonly StoredListingRow[]> {
  const [first] = args.group;
  if (first === undefined) {
    return [];
  }
  const base = args.tables.client.from(args.tables.listingsTable).select(SELECT_COLUMNS);
  const query =
    first.access_kind === "api_resource"
      ? base
          .eq("access_kind", "api_resource")
          .eq("api_service", first.api_service ?? "")
          .eq("api_base_url", first.api_base_url ?? "")
          .in("external_dataset_id", _unique(args.group.map((row) => row.external_dataset_id)))
      : base
          .eq("access_kind", "pipeline_parquet")
          .eq("pipeline_name", first.pipeline_name)
          .in("parquet_file_name", _unique(args.group.map((row) => row.parquet_file_name)));
  const { data, error } = await query;
  _throwOnError({ error, action: `reading rows from ${args.tables.listingsTable}` });
  return (data ?? []) as unknown as StoredListingRow[];
}

async function _readRowsMatching(args: {
  tables: _Tables;
  rows: readonly ListingRowValues[];
}): Promise<readonly StoredListingRow[]> {
  // An API dataset's resources share an `external_dataset_id`, so a group
  // reads every resource of each dataset and the key filter keeps only the
  // ones asked for.
  const groups = _groupBy(args.rows, (row) => {
    return row.access_kind === "api_resource"
      ? JSON.stringify([row.access_kind, row.api_service, row.api_base_url])
      : JSON.stringify([row.access_kind, row.pipeline_name]);
  });
  const stored = await Promise.all(
    [...groups.values()].map((group) => {
      return _readGroup({ tables: args.tables, group });
    }),
  );
  const wantedKeys = new Set(args.rows.map(buildListingKeyFromRow));
  return stored.flat().filter((row) => {
    return wantedKeys.has(buildListingKeyFromRow(row));
  });
}

async function _upsertKind(args: {
  tables: _Tables;
  rows: readonly ListingRowValues[];
  onConflict: string;
}): Promise<readonly StoredListingRow[]> {
  if (args.rows.length === 0) {
    return [];
  }
  const { data, error } = await args.tables.client
    .from(args.tables.listingsTable)
    .upsert(
      args.rows.map((row) => {
        return { ...row };
      }),
      { onConflict: args.onConflict },
    )
    .select(SELECT_COLUMNS);
  _throwOnError({ error, action: `upserting rows into ${args.tables.listingsTable}` });
  return (data ?? []) as unknown as StoredListingRow[];
}

async function _upsertRows(args: {
  tables: _Tables;
  rows: readonly ListingRowValues[];
}): Promise<readonly StoredListingRow[]> {
  const apiRows = await _upsertKind({
    tables: args.tables,
    rows: args.rows.filter((row) => {
      return row.access_kind === "api_resource";
    }),
    onConflict: API_RESOURCE_CONFLICT,
  });
  const parquetRows = await _upsertKind({
    tables: args.tables,
    rows: args.rows.filter((row) => {
      return row.access_kind === "pipeline_parquet";
    }),
    onConflict: PIPELINE_PARQUET_CONFLICT,
  });
  return [...apiRows, ...parquetRows];
}

async function _replaceRelationColumns(args: {
  tables: _Tables;
  listingId: string;
  columns: readonly RelationColumnRow[];
}): Promise<void> {
  const { client, relationColumnsTable } = args.tables;
  const { error: deleteError } = await client
    .from(relationColumnsTable)
    .delete()
    .eq("catalog_listing_id", args.listingId);
  _throwOnError({ error: deleteError, action: `deleting columns from ${relationColumnsTable}` });
  if (args.columns.length === 0) {
    return;
  }
  const { error: insertError } = await client.from(relationColumnsTable).insert(
    args.columns.map((column) => {
      return { ...column, catalog_listing_id: args.listingId };
    }),
  );
  _throwOnError({ error: insertError, action: `writing columns to ${relationColumnsTable}` });
}

/**
 * Creates a listings store over two Supabase tables that satisfy the
 * engine's output contract (see the README). Every method throws with the
 * table name and Supabase's message when a request fails.
 *
 * @param tables.client - A service-role client, since listings bypass RLS.
 * @param tables.listingsTable - The listings table.
 * @param tables.relationColumnsTable - The relation columns table.
 * @returns The store.
 */
export function createSupabaseListingsStore(tables: _Tables): ListingsStore {
  return {
    readHighWaterMark: (args) => {
      return _readHighWaterMark({ ...tables, ...args });
    },
    readRowsMatching: (rows) => {
      return _readRowsMatching({ tables, rows });
    },
    upsertRows: (rows) => {
      return _upsertRows({ tables, rows });
    },
    replaceRelationColumns: (args) => {
      return _replaceRelationColumns({ tables, ...args });
    },
  };
}
```

```ts
/** Every drive of a run failed. Carries the run's report. */
export class EtlRunFailedError extends Error {
  /** The run's report, with each drive's error. */
  readonly report: EtlRunReport;

  constructor(report: EtlRunReport) {
    const failures = report.drives
      .map((drive) => {
        return `${drive.driveKey}: ${drive.error ?? "unknown error"}`;
      })
      .join("; ");
    super(`Every drive failed. ${failures}`);
    this.name = "EtlRunFailedError";
    this.report = report;
  }
}
```

```ts
async function _resolveDriveWindow<TDrive, TConnection>(
  args: _DriveArgs<TDrive, TConnection>,
): Promise<EtlRunWindow> {
  const { config, options } = args;
  if (options.mode !== "incremental" || options.since !== undefined) {
    return resolveRunWindow({ options, highWaterMark: undefined });
  }
  const readDefaultHighWaterMark = () => {
    return args.stores.listings.readHighWaterMark({
      pipelineName: config.name,
      apiBaseUrl: args.driveKey,
    });
  };
  const readCustom = config.incremental?.readHighWaterMark;
  const highWaterMark =
    readCustom === undefined
      ? await readDefaultHighWaterMark()
      : await readCustom({ drive: args.drive, driveKey: args.driveKey, readDefaultHighWaterMark });
  return resolveRunWindow({ options, highWaterMark });
}

  const getConnectionTimestamp = config.incremental?.getConnectionTimestamp;
  const { connections, state } = selectPageConnections({
    page: args.page,
    driveKey: args.driveKey,
    window: args.window,
    state: progress.selection,
    getConnectionId: config.getConnectionId,
    getConnectionTimestamp,
  });
  const relations = await promiseMapSequential(connections, (connection) => {
    return config.transform({ drive: args.drive, connection, pipelineRunId: args.pipelineRunId });
  });
  const loaded = await loadRelationsPage({
    relations: relations.flat(),
    context: {
      pipelineName: config.name,
      pipelineRunId: args.pipelineRunId,
      service: config.service,
      driveKey: args.driveKey,
      syncedAt: args.syncedAt,
      stores: args.stores,
      bucket: config.output.bucket,
    },
  });
  const newest = connections.at(-1);
  return {
    connectionsFetched: progress.connectionsFetched + connections.length,
    relationsWritten: progress.relationsWritten + loaded.written,
    relationsUnchanged: progress.relationsUnchanged + loaded.unchanged,
    newestTimestamp:
      newest === undefined || getConnectionTimestamp === undefined
        ? progress.newestTimestamp
        : getConnectionTimestamp(newest),
    selection: state,
  };
}

async function _crawlPages<TDrive, TConnection>(
  args: _PagesArgs<TDrive, TConnection>,
): Promise<Readonly<{ progress: _DriveProgress; error?: unknown }>> {
  let progress = _createProgress(args.remaining);
  try {
    const pages = args.config.extract({
      drive: args.drive,
      window: args.window,
      pipelineRunId: args.pipelineRunId,
    });
    // Sequential on purpose: each page is written before the next is
    // fetched, so the listings table always holds an oldest-first prefix of
    // the drive. That is what makes the high-water mark safe to resume from.
    for await (const page of pages) {
      progress = await _processPage({ ...args, page, progress });
      if (progress.selection.remaining <= 0) {
        break;
      }
    }
    return { progress };
  } catch (error: unknown) {
    return { progress, error };
  }
}

function _buildDriveReport(args: {
  driveKey: string;
  window: EtlRunWindow;
  progress: _DriveProgress;
  error: unknown;
}): EtlDriveReport {
  const { progress, error } = args;
  const base = {
    driveKey: args.driveKey,
    window: args.window,
    connectionsFetched: progress.connectionsFetched,
    relationsWritten: progress.relationsWritten,
    relationsUnchanged: progress.relationsUnchanged,
    newestTimestamp: progress.newestTimestamp,
  };
  if (error === undefined) {
    return { ...base, status: "ok" };
  }
  return {
    ...base,
    status: "failed",
    error: error instanceof Error ? error.message : String(error),
  };
}

/**
 * Crawls one drive: resolves its window, then fetches, transforms, and loads
 * one page at a time until the source runs out or the run's limit is
 * reached. A failure never escapes: the drive is reported `failed`, with the
 * counts of what it wrote before the failure.
 *
 * @returns The drive's report.
 */
export async function crawlDrive<TDrive, TConnection>(
  args: CrawlDriveArgs<TDrive, TConnection>,
): Promise<EtlDriveReport> {
  const driveKey = args.config.getDriveKey(args.drive);
  try {
    const window = await _resolveDriveWindow({ ...args, driveKey });
    const { progress, error } = await _crawlPages({ ...args, driveKey, window });
    return _buildDriveReport({ driveKey, window, progress, error });
  } catch (error: unknown) {
    return _buildDriveReport({
      driveKey,
      window: { since: args.options.since, until: args.options.until },
      progress: _createProgress(args.remaining),
      error,
    });
  }
}
```

```ts
async function _crawlDrivesInOrder<TDrive, TConnection>(args: {
  config: EtlEngineConfig<TDrive, TConnection>;
  drives: readonly TDrive[];
  options: EtlRunOptions;
  pipelineRunId: EtlPipelineRunId;
  stores: EtlStores;
  syncedAt: Date;
}): Promise<readonly EtlDriveReport[]> {
  const { config, options } = args;
  const limit = options.limit ?? Number.POSITIVE_INFINITY;
  return promiseReduce(
    args.drives,
    async (reports: readonly EtlDriveReport[], drive) => {
      const fetched = reports.reduce((sum, report) => {
        return sum + report.connectionsFetched;
      }, 0);
      const remaining = limit - fetched;
      const report =
        remaining <= 0
          ? _buildSkippedReport({ driveKey: config.getDriveKey(drive), options })
          : await crawlDrive({
              config,
              drive,
              options,
              pipelineRunId: args.pipelineRunId,
              stores: args.stores,
              remaining,
              syncedAt: args.syncedAt,
            });
      return [...reports, report];
    },
    [] as readonly EtlDriveReport[],
  );
}

/**
 * Creates one pipeline's engine. `run()` crawls every drive in
 * `listDrives` order, one page at a time, and loads each page into the
 * configured listings table and bucket before fetching the next. See the
 * package README for the output contract and run modes.
 *
 * @param config - The pipeline's drives, extract, transform, and output.
 * @returns The engine.
 */
export function createEtlEngine<TDrive, TConnection>(
  config: EtlEngineConfig<TDrive, TConnection>,
): EtlEngineInstance {
  return {
    run: async (options) => {
      _assertCanRun({ config, options });
      const stores = config.stores ?? _createSupabaseStores(config);
      const pipelineRunId = randomUUID() as EtlPipelineRunId;
      const drives = await config.listDrives();
      const driveReports = await _crawlDrivesInOrder({
        config,
        drives,
        options,
        pipelineRunId,
        stores,
        syncedAt: new Date(),
      });
      const report: EtlRunReport = { pipelineRunId, mode: options.mode, drives: driveReports };
      const isEveryDriveFailed =
        driveReports.length > 0 &&
        driveReports.every((drive) => {
          return drive.status === "failed";
        });
      if (isEveryDriveFailed) {
        throw new EtlRunFailedError(report);
      }
      return report;
    },
  };
}
```

```ts
 * @returns The coverage, or `undefined` when the CSV has no `Year` column or

  db: NodeDuckDb;
  csvPath: string;
  columnNames: readonly string[];
}): Promise<WdiYearCoverage | undefined> {
  const yearColumn = args.columnNames.find((name) => {
    return name.toLowerCase() === "year";
  });
  if (yearColumn === undefined) {
    return undefined;
  }
```

```ts
async function* _extractWdiCsvFiles(
  pipelineRunId: EtlPipelineRunId,
): AsyncGenerator<readonly WdiCsvFile[]> {
  const inputDir = getEtlPipelineInputDir(WDI_PIPELINE_NAME);
  await mkdir(inputDir, { recursive: true });
  const csvFilenames = (await readdir(inputDir))
    .filter((name) => {
      return name.toLowerCase().endsWith(".csv");
    })
    .sort();
  if (csvFilenames.length === 0) {
    throw new Error(`No CSV files found under ${inputDir}. Add one or more .csv files.`);
  }
  await Promise.all(
    csvFilenames.map((csvFilename) => {
      return EtlEngine.storeExtractedData({
        pipelineName: WDI_PIPELINE_NAME,
        pipelineRunId,
        sourcePath: join(inputDir, csvFilename),
        destinationBasename: csvFilename,
      });

  );
  const extractDir = getEtlOutputDir(WDI_PIPELINE_NAME, pipelineRunId, "extract");
  yield csvFilenames.map((csvFilename) => {
    return { csvFilename, csvPath: join(extractDir, csvFilename) };
  });
}

async function _transformWdiCsvFile(file: WdiCsvFile): Promise<PipelineParquetRelation> {
  const db = new NodeDuckDb();
  try {
    const csvColumns = await db.sniffCsv({ csvPath: file.csvPath });

      tableBaseName: file.csvFilename.replace(/\.csv$/i, ""),
      csvPath: file.csvPath,
```

````markdown
## Run modes

| Options | Each drive's window |
| --- | --- |
| `{ mode: "full" }` | No lower bound: every connection is fetched |
| `{ mode: "incremental" }` | From the drive's high-water mark |
| `{ since }` | From `since` (inclusive), in either mode |
| `{ until }` | Adds an exclusive upper bound |
| `{ limit }` | Stops after this many connections across all drives |

Both modes write only rows that changed. A row is unchanged when every
column the engine writes, except `pipeline_run_id` and `date_of_last_sync`,
equals the stored value.

**Resuming.** With `incremental` config, connections must arrive oldest
first, and each page is written before the next is fetched. The listings
table therefore always holds an oldest-first prefix of each drive, and an
incremental run resumes from it after a cancelled run, a limited run, or a
finished one. The lower bound is inclusive, so connections that share the
last stored timestamp are fetched again and skipped if unchanged. A source
that returns an older connection after a newer one fails that drive with
`OutOfOrderConnectionError`.

**The high-water mark** defaults to `max(date_of_last_update)` for the
pipeline's rows with `api_base_url` equal to the drive key. Override it with
`incremental.readHighWaterMark`.

**Failures.** A failing drive is reported `failed` and the next drive still
runs. `run()` rejects with `EtlRunFailedError` only when every drive fails.

## Output contract

The engine writes only these listings-table columns
(`LISTING_CONTRACT_COLUMNS`), and needs a table that has them:

| Column | Written for |
| --- | --- |
| `id` (read only), `access_kind`, `pipeline_name`, `pipeline_run_id`, `display_name`, `external_organization_name`, `external_dataset_id`, `date_of_last_update`, `date_of_last_sync` | every row |
| `external_service_name`, `source_url`, `canonical_urls`, `license`, `update_frequency`, `description`, `notes`, `metadata`, `coverage_start_date`, `coverage_end_date` | every row, `null` when unset |
| `api_service`, `api_base_url`, `api_resource_id`, `api_resource_format` | `api_resource` rows |
| `parquet_file_name` | `pipeline_parquet` rows |

Conflict targets: `(api_service, api_base_url, external_dataset_id,
api_resource_id)` and `(parquet_file_name, pipeline_name)`. Any other column,
such as an app-owned cache, is never written.

The relation columns table needs `catalog_listing_id`, `column_name`,
`display_order`, `original_data_type`, and `cast_data_type`. Writing a row
replaces that listing's whole column set.

**Parquet relations** are converted from CSV to ZSTD Parquet, uploaded to
`<bucket>/<pipeline name>/datasets/<table>.parquet` unless the stored row's
`metadata.parquet_sha256` matches, and written with these metadata keys:
`open_data_bucket`, `storage_object_path`, `table` (`name`, `row_count`,
`column_names`), and `parquet_sha256`. The object's public URL is appended to
`canonical_urls`, and the relation columns come from the Parquet file.
````

```ts
/** A trailing `Z` or `+hh:mm` offset. */
const ZONE_SUFFIX = /(Z|[+-]\d{2}:?\d{2})$/;

/**
 * Reads a CKAN timestamp. CKAN writes `metadata_modified` as naive UTC with
 * microseconds (`2026-10-20T13:47:48.858245`), which `Date` would read as
 * local time, so a value with no zone is read as UTC. The fraction is cut to
 * milliseconds, which is also the precision CKAN's search index keeps.
 *
 * @throws When the value is not a timestamp.
 */
export function parseCkanTimestamp(value: string): Date {
  const milliseconds = value.replace(/(\.\d{3})\d+/, "$1");
  const date = new Date(
    ZONE_SUFFIX.test(milliseconds) ? milliseconds : `${milliseconds}Z`,
  );
  if (Number.isNaN(date.getTime())) {
    throw new Error(`Not a CKAN timestamp: ${JSON.stringify(value)}.`);
  }
  return date;
}
```

```ts
import type { Mixin } from "@avandar/modules";
import type {
  HttpClientState,
  HttpMiddleware,
  HttpMiddlewareRequest,
} from "$/utils/HttpClientFactory/HttpClientFactory.types.ts";

type _RetryBase = {
  getMiddlewares(): HttpClientState["middlewares"];
};

type _RetryOptions = Readonly<{
  maxRetries: number;
  initialDelayMs: number;
  sleep: (ms: number) => Promise<void>;
}>;

function _sleep(ms: number): Promise<void> {
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

async function _sendWithRetries(args: {
  request: Readonly<HttpMiddlewareRequest>;
  next: (request: Readonly<HttpMiddlewareRequest>) => Promise<Response>;
  options: _RetryOptions;
  attempt: number;
}): Promise<Response> {
  const response = await args.next(args.request);
  if (response.status !== 429 || args.attempt >= args.options.maxRetries) {
    return response;
  }
  await response.body?.cancel();
  await args.options.sleep(args.options.initialDelayMs * 2 ** args.attempt);
  return _sendWithRetries({ ...args, attempt: args.attempt + 1 });
}

/**
 * Retries a request answered with HTTP 429, waiting `initialDelayMs` and
 * doubling the wait after each retry. For APIs that rate-limit without a
 * `Retry-After` header. Any other status passes through untouched.
 *
 * Apply it before `withRateLimit`, so each retry is scheduled through the
 * rate limiter like any other request.
 *
 * @param options.maxRetries - Retries after the first attempt.
 * @param options.initialDelayMs - The wait before the first retry.
 * @param options.sleep - Replaces the timer, for tests.
 * @returns The mixin.
 */
export function withRetryOnTooManyRequests(options: {
  maxRetries: number;
  initialDelayMs: number;
  sleep?: (ms: number) => Promise<void>;
}): Mixin<_RetryBase, Pick<HttpClientState, "middlewares">> {
  const retryOptions: _RetryOptions = {
    maxRetries: options.maxRetries,
    initialDelayMs: options.initialDelayMs,
    sleep: options.sleep ?? _sleep,
  };
  const retryMiddleware: HttpMiddleware = (request, next) => {
    return _sendWithRetries({ request, next, options: retryOptions, attempt: 0 });
  };
  return (base) => {
    return {
      state: {
        middlewares: [...(base.getMiddlewares() ?? []), retryMiddleware],
      },
    };
  };
}
```

```ts
  }
  const date =
    typeof value === "string" && ISO_TIMESTAMP_WITH_ZONE.test(value)
      ? new Date(value)
      : undefined;
  if (date === undefined || Number.isNaN(date.getTime())) {
    throw new Error(
      `--${name} must be an ISO 8601 timestamp with a timezone, such as ` +
        `2026-10-01T00:00:00Z. Got ${JSON.stringify(value)}.`,
    );
  }
  return date;
}

function _parseLimit(value: unknown): number | undefined {
  if (value === undefined) {
    return undefined;
  }
  const limit = typeof value === "string" && value.trim() !== "" ? Number(value) : value;
  if (typeof limit !== "number" || !Number.isInteger(limit) || limit < 1) {
    throw new Error(`A pipeline limit must be a positive integer, got ${JSON.stringify(value)}.`);
  }
  return limit;
}

/**
 * Validates a run's options, from a request body or from argv. A full run is
 * the default. `since` and `until` must carry a timezone, so a bare date is
 * never guessed at.
 *
 * @throws When an option is malformed, when `incremental` and `since` are
 * both given (they name two different lower bounds), or when `since` is not
 * before `until`.
 */
export function parsePipelineRunOptions(raw: RawPipelineRunOptions): PipelineRunOptions {
  const isIncremental = _parseIncremental(raw.incremental);
  const since = _parseTimestamp("since", raw.since);
  const until = _parseTimestamp("until", raw.until);
  if (isIncremental && since !== undefined) {
    throw new Error(
      "Pass either --incremental or --since, not both: they name two different lower bounds.",
    );
  }
  if (since !== undefined && until !== undefined && since.getTime() >= until.getTime()) {
    throw new Error("--since must be earlier than --until.");
  }
  return {
    mode: isIncremental ? "incremental" : "full",
    ...(since === undefined ? {} : { since }),
    ...(until === undefined ? {} : { until }),
    ...(raw.limit === undefined ? {} : { limit: _parseLimit(raw.limit) }),
  };
}
```

```ts
import type { EtlRunWindow } from "@avandar/etl";
import type {
  SocrataDiscoveryClient,
  SocrataDiscoveryResult,
} from "$/connectors/open-data/SocrataDiscoveryClient/SocrataDiscoveryClient.types";
import type { ConfiguredDrive } from "@pipelines/socrata/configuredDrives/configuredDrives";

/** One Discovery API page. */
export const SOCRATA_PAGE_SIZE = 1000;

/**
 * How far each page reaches back into the one before it. A dataset modified
 * mid-crawl moves to the end of the ascending order and shifts every later
 * result one place earlier, which would skip a result at a page boundary.
 * The engine drops the repeats this causes.
 */
export const SOCRATA_PAGE_OVERLAP = 20;

/** The Discovery API rejects any request whose offset plus limit is above this. */
export const SOCRATA_MAX_RESULTS = 10_000;

function _isBefore(result: SocrataDiscoveryResult, date: Date | undefined): boolean {
  return date !== undefined && Date.parse(result.updatedAt) < date.getTime();
}

function _takeBeforeUntil(
  results: readonly SocrataDiscoveryResult[],
  until: Date | undefined,
): SocrataDiscoveryResult[] {
  const firstAtOrAfter = results.findIndex((result) => {
    return until !== undefined && !_isBefore(result, until);
  });
  return firstAtOrAfter === -1 ? [...results] : results.slice(0, firstAtOrAfter);
}

async function* _pagesFrom(args: {
  drive: Readonly<ConfiguredDrive>;
  window: EtlRunWindow;
  discovery: SocrataDiscoveryClient;
  offset: number;
}): AsyncGenerator<readonly SocrataDiscoveryResult[]> {
  const limit = Math.min(SOCRATA_PAGE_SIZE, SOCRATA_MAX_RESULTS - args.offset);
  const { results, resultSetSize } = await args.discovery.search({
    domains: [new URL(args.drive.baseUrl).host],
    order: "updatedAt ASC",
    limit,
    offset: args.offset,
  });
  if (resultSetSize > SOCRATA_MAX_RESULTS) {
    throw new Error(
      `Socrata drive ${args.drive.baseUrl} has ${String(resultSetSize)} results, over ` +
        `the Discovery API's ${String(SOCRATA_MAX_RESULTS)}-result paging limit.`,
    );
  }
  const afterSince = results.filter((result) => {
    return !_isBefore(result, args.window.since);
  });
  const inWindow = _takeBeforeUntil(afterSince, args.window.until);
  yield inWindow;
  const hasReachedUntil = inWindow.length < afterSince.length;
  const isLastPage =
    results.length < limit || args.offset + results.length >= resultSetSize;
  if (hasReachedUntil || isLastPage) {
    return;
  }
  yield* _pagesFrom({
    ...args,
    offset: args.offset + results.length - SOCRATA_PAGE_OVERLAP,
  });
}

/**
 * Pages through one Socrata drive's datasets, least recently updated first.
 * The Discovery API has no date filter, so the window is applied here: a
 * result before `since` is skipped, and the crawl stops at the first result
 * at or after `until`.
 *
 * @throws When the drive has more results than the Discovery API can page.
 */
export function extractSocrataDrive(args: {
  drive: Readonly<ConfiguredDrive>;
  window: EtlRunWindow;
  discovery: SocrataDiscoveryClient;
}): AsyncGenerator<readonly SocrataDiscoveryResult[]> {
  return _pagesFrom({ ...args, offset: 0 });
}
```

```ts
import { getHdxDatasetTimestamp } from "@pipelines/hdx/getHdxDatasetTimestamp/getHdxDatasetTimestamp";
import type { EtlRunWindow } from "@avandar/etl";
import type { CkanPackage } from "$/connectors/open-data/CkanClient/CkanClient.types";
import type {
  CkanDatasetSearchCursor,
  CkanDiscoveryClient,
} from "$/connectors/open-data/CkanDiscoveryClient/CkanDiscoveryClient.types";
import type { HdxDrive } from "@pipelines/hdx/hdxDrive";

/** CKAN's largest `package_search` page. */
export const HDX_PAGE_SIZE = 1000;

type _PageArgs = Readonly<{
  drive: Readonly<HdxDrive>;
  window: EtlRunWindow;
  discovery: CkanDiscoveryClient;
  pageSize: number;
  cursor: CkanDatasetSearchCursor | undefined;
}>;

function _isAfterCursor(
  ckanPackage: CkanPackage,
  cursor: CkanDatasetSearchCursor | undefined,
): boolean {
  if (cursor === undefined) {
    return true;
  }
  const time = getHdxDatasetTimestamp(ckanPackage).getTime();
  const cursorTime = cursor.modifiedAt.getTime();
  return time > cursorTime || (time === cursorTime && ckanPackage.name > cursor.name);
}

async function* _pagesAfter(args: _PageArgs): AsyncGenerator<readonly CkanPackage[]> {
  const page = await args.discovery.searchDatasets({
    baseUrl: args.drive.baseUrl,
    modifiedSince: args.window.since,
    modifiedBefore: args.window.until,
    after: args.cursor,
    rows: args.pageSize,
  });
  const fresh = page.filter((ckanPackage) => {
    return _isAfterCursor(ckanPackage, args.cursor);
  });
  if (page.length === args.pageSize && fresh.length === 0) {
    throw new Error(
      `HDX returned a full page with nothing after ${args.cursor?.name ?? "the start"}; ` +
        "stopping instead of requesting it again.",
    );
  }
  yield fresh;
  const last = page.at(-1);
  if (page.length < args.pageSize || last === undefined) {
    return;
  }
  yield* _pagesAfter({
    ...args,
    cursor: { modifiedAt: getHdxDatasetTimestamp(last), name: last.name },
  });
}

/**
 * Pages through one HDX drive's datasets, least recently modified first,
 * inside the window. Each page after the first asks for the datasets
 * strictly after the previous page's last `(metadata_modified, name)` pair,
 * so a dataset modified mid-crawl is neither skipped nor repeated.
 *
 * @throws When a full page has nothing after the cursor, which would
 * otherwise request the same page forever.
 */
export function extractHdxDatasets(args: {
  drive: Readonly<HdxDrive>;
  window: EtlRunWindow;
  discovery: CkanDiscoveryClient;
  pageSize?: number;
}): AsyncGenerator<readonly CkanPackage[]> {
  return _pagesAfter({
    drive: args.drive,
    window: args.window,
    discovery: args.discovery,
    pageSize: args.pageSize ?? HDX_PAGE_SIZE,
    cursor: undefined,
  });
}
```

```ts
    name: "hdx",

    transform: ({ drive: current, connection }) => {
      return makeApiResourceRelationsFromCkanPackage({ drive: current, ckanPackage: connection });
    },
    incremental: { getConnectionTimestamp: getHdxDatasetTimestamp },
```

```ts
import type { PipelineRunReport } from "@ava-cli/PipelineCli/PipelineRunReport/PipelineRunReport.schemas";

type _DriveReport = PipelineRunReport["drives"][number];

function _formatDriveLine(drive: _DriveReport): string {
  const since = drive.window.since ?? "the start";
  const until = drive.window.until ?? "now";
  const counts =
    `fetched ${String(drive.connectionsFetched)}, ` +
    `written ${String(drive.relationsWritten)}, ` +
    `unchanged ${String(drive.relationsUnchanged)}`;
  return `${drive.status.padEnd(7)} ${drive.driveKey}  ${since} to ${until}  ${counts}, newest ${drive.newestTimestamp ?? "none"}`;
}

/** Prints a run's id and mode, then one line per drive. */
export function printPipelineRunReport(report: PipelineRunReport): void {
  printSuccess(`Pipeline run ${report.pipelineRunId} (${report.mode})`);
  if (report.drives.length === 0) {
    printInfo("No drives are configured, so nothing was crawled.");
    return;
  }
  report.drives.forEach((drive) => {
    if (drive.status === "failed") {
      printError(`${_formatDriveLine(drive)}\n        error: ${drive.error ?? "unknown error"}`);
      return;
    }
    if (drive.status === "skipped") {
      printWarn(`${_formatDriveLine(drive)} (limit reached)`);
      return;
    }
    printInfo(_formatDriveLine(drive));
  });
}
```

```markdown
it writes `data_catalog_listings` rows and relation columns, and Parquet
objects for `world-bank__wdi`, writing only rows that changed.

`hdx` and `socrata` crawl each drive oldest-modified first (`hdx` by
`metadata_modified` with `package_search` keyset paging, `socrata` by
`updatedAt`) and save each page before fetching the next. `ava pipeline run
<name> --incremental` therefore resumes each drive from the newest
`date_of_last_update` it already holds, whether the last run finished, was
cancelled, or stopped at `--limit`. See the
[engine spec](../superpowers/specs/2026-10-20-etl-engine-supabase-load-and-incremental-runs-design.md).

Both send every request through an `HttpClientFactory` client with
`withRateLimit({ minTimeMs: 1000 })`. HDX also retries an HTTP 429 three
times with exponential backoff, since it sends no `Retry-After`. The Socrata
pipeline shares one client across all of its drives, because every drive is
read through the same Discovery API host.
```

